if(CMAKE_BUILD_EARLY_EXPANSION)
  idf_component_register(REQUIRES esp_common)
  return()
endif()
set(DAEMON_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../..")
include(FetchContent)
set(LVGL_ARCHIVE "${DAEMON_ROOT}/.cache/downloads/lvgl-v9.2.2.tar.gz")
if(NOT EXISTS "${LVGL_ARCHIVE}")
  set(LVGL_ARCHIVE "https://codeload.github.com/lvgl/lvgl/tar.gz/refs/tags/v9.2.2")
endif()
FetchContent_Declare(daemon_lvgl URL "${LVGL_ARCHIVE}"
  URL_HASH SHA256=129b4e00e06639fa79d7e8a6cab3c1ecce2445b1a246652ccd34f22e7b17ad6f
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_GetProperties(daemon_lvgl)
if(NOT daemon_lvgl_POPULATED)
  FetchContent_Populate(daemon_lvgl)
endif()
file(GLOB_RECURSE LVGL_SOURCES "${daemon_lvgl_SOURCE_DIR}/src/*.c")
idf_component_register(SRCS ${LVGL_SOURCES} INCLUDE_DIRS "${daemon_lvgl_SOURCE_DIR}" "${DAEMON_ROOT}/presentation" REQUIRES esp_common)
target_compile_definitions(${COMPONENT_LIB} PUBLIC LV_CONF_INCLUDE_SIMPLE=1)
